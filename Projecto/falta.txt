using System;
using System.IO;

class Program
{
    static void Main()
    {
        // Ruta del archivo de texto
        string filePath = "Sucursal.txt";

        // Lee todas las líneas del archivo
        string[] lines = File.ReadAllLines(filePath);

        // Itera a través de cada línea
        foreach (string line in lines)
        {
            // Divide la línea en partes utilizando un separador (por ejemplo, espacio en blanco)
            string[] parts = line.Split(",");

            // Itera a través de cada parte e imprímela
            foreach (string part in parts)
            {
                Console.WriteLine(part);
            }
        }
    }
}
